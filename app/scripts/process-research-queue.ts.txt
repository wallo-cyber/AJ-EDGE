import { createClient } from '@supabase/supabase-js';
import { ResilientResearchEngine } from '../lib/research/resilient-engine';

const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL!;
const supabaseServiceKey = process.env.SUPABASE_SECRET_KEY!;

if (!supabaseUrl || !supabaseServiceKey) {
  console.error('Error: NEXT_PUBLIC_SUPABASE_URL or SUPABASE_SECRET_KEY is missing in environment variables.');
  process.exit(1);
}

const supabase = createClient(supabaseUrl, supabaseServiceKey);

async function processQueueInBatches(batchSize = 10) {
  const engine = new ResilientResearchEngine();

  console.log(`[QueueWorker] Fetching up to ${batchSize} pending research jobs...`);

  const { data: jobs, error: fetchError } = await supabase
    .from('agent_jobs')
    .select('*')
    .eq('job_type', 'external_research')
    .in('status', ['pending', 'queued'])
    .order('created_at', { ascending: true })
    .limit(batchSize);

  if (fetchError) {
    console.error('[QueueWorker] Failed to fetch jobs:', fetchError.message);
    return;
  }

  if (!jobs || jobs.length === 0) {
    console.log('[QueueWorker] Queue is clean. No pending research jobs.');
    return;
  }

  console.log(`[QueueWorker] Found ${jobs.length} jobs to process.`);

  for (const job of jobs) {
    const companyName = job.payload?.company_name || job.payload?.name;

    if (!companyName) {
      await supabase
        .from('agent_jobs')
        .update({ status: 'cancelled', error_details: 'Missing company name in payload' })
        .eq('id', job.id);
      continue;
    }

    await supabase
      .from('agent_jobs')
      .update({ 
        status: 'running', 
        started_at: new Date().toISOString(),
        attempts: (job.attempts || 0) + 1 
      })
      .eq('id', job.id);

    try {
      const { results, providerUsed } = await engine.executeResearch(companyName);

      await supabase.from('agent_results').insert({
        job_id: job.id,
        company_id: job.payload?.company_id,
        result_data: results,
        metadata: { provider_used: providerUsed, processed_at: new Date().toISOString() },
      });

      await supabase
        .from('agent_jobs')
        .update({ 
          status: 'completed', 
          completed_at: new Date().toISOString(),
          error_details: null 
        })
        .eq('id', job.id);

      console.log(`[QueueWorker] ✅ Job ${job.id} (${companyName}) completed via [${providerUsed}]`);

    } catch (err: any) {
      console.error(`[QueueWorker] ❌ Job ${job.id} failed: ${err.message}`);

      const maxAttempts = 3;
      const nextStatus = (job.attempts || 0) + 1 >= maxAttempts ? 'failed' : 'pending';

      await supabase
        .from('agent_jobs')
        .update({
          status: nextStatus,
          error_details: err.message,
          updated_at: new Date().toISOString(),
        })
        .eq('id', job.id);
    }
  }
}

processQueueInBatches(15);
